Messung: was eine nur lesende SQLite-Verbindung neben der Datenbankdatei schreibt
Arbeitspaket 261023-1305-sqlite-datenbanken-im-dateifenster-und-in-der-vorschau-durchsehen, Plan 261023-1507, Schritt 1
Grundlage: Bericht 261023-1324-sqlite-datenbanken-durchsehen-machbarkeit-und-zuschnitt.md, Abschnitt 3
Gefahren: 261023-1554 auf diesem Geraet (Darwin 24.6.0), SQLite 3.43.2 des Systems

Gemessen wurde ausschliesslich an Datenbanken, die das Skript selbst in einem frischen
Wegwerfordner unter dem Scratchpad der Sitzung anlegt und am Ende loescht; jede Datenbank
liegt in einem eigenen Ordner, in dem sonst nichts steht, damit "Ordner:" jede neu
entstandene Datei zeigt. wordstore.db und der Baum sind nicht beruehrt.

Abweichung vom Plan, in (a) begruendet: /usr/bin/sqlite3 laedt /usr/lib/libsqlite3.dylib
NICHT (otool -L fuehrt sie nicht; das Programm traegt SQLite statisch). Damit die Messung
trotzdem die Bibliothek trifft, die Schritt 2 bindet, faehrt das Skript (b) bis (e) zweimal:
mit /usr/bin/sqlite3 ("cli") und mit einem Wegwerf-Leser in C ("lib", leser.c unten),
gelinkt gegen /usr/lib/libsqlite3.dylib und geoeffnet mit
sqlite3_open_v2(uri, SQLITE_OPEN_READONLY | SQLITE_OPEN_URI), also dem Weg aus Schritt 2.
Beide melden dieselbe Fassung und dieselbe sqlite3_sourceid; beide Laeufe ergeben dasselbe.

Weitere Abweichungen:
- (c): der Schreiber endet mit kill -9, ohne zu schliessen. Ein sauber schliessender
  letzter Schreiber faehrt selbst einen Checkpoint; -wal mit einer nicht zurueckgeschriebenen
  Zeile bleibt nur nach einem Abbruch stehen, und genau den Fall verlangt (c).
- (d): das System-SQLite laesst nach dem sauberen Schliessen einer WAL-Datenbank ein leeres
  -wal (0 Bytes) und ein -shm stehen. Fuer "ohne -wal und -shm" entfernt das Skript beide
  nach dem Anlegen; das ist der Zustand einer kopierten oder heruntergeladenen Datei.
- Jede Messung "vor" steht neben einer Kontrollmessung ohne Leser im Abstand von 1 s, damit
  eine Aenderung dem Leser und nicht dem ruhenden Schreiber zugerechnet wird.
- %m in stat zaehlt ganze Sekunden; entscheidend ist die Pruefsumme. Ein kurzlebiges
  -journal, das waehrend des Lesens entstuende und wieder verschwaende, sieht die Messung
  nach dem Lesen nicht; sie sieht nur, dass danach keins steht und die Hauptdatei gleich ist.

==================================================================== Skript: messen.sh
#!/bin/bash
# Misst, was eine nur lesende SQLite-Verbindung neben einer Datenbankdatei schreibt.
# Jede Datenbank legt dieses Skript selbst an, jede in einem eigenen frischen Ordner,
# in dem sonst nichts liegt. Gelesen wird mit zwei Lesern:
#   cli = /usr/bin/sqlite3 mit file:...?<parameter>
#   lib = ein Wegwerf-Leser (leser.c), gelinkt gegen /usr/lib/libsqlite3.dylib,
#         sqlite3_open_v2(SQLITE_OPEN_READONLY|SQLITE_OPEN_URI), count(*), sqlite3_close_v2
set -u
SQ=/usr/bin/sqlite3
HIER=$(cd "$(dirname "$0")" && pwd)
W=$(mktemp -d "${TMPDIR:-/tmp}/krk-shm.XXXXXX")
H=$(mktemp -d "${TMPDIR:-/tmp}/krk-shm-hilfe.XXXXXX")   # Hilfsdateien, nie neben einer Datenbank
cc -o "$H/leser" "$HIER/leser.c" -lsqlite3 || exit 1
echo "Wegwerfordner: $W"

zustand() { # $1 = Beschriftung, $2 = Datenbankpfad
  echo "--- $1"
  for s in "" -wal -shm -journal; do
    f="$2$s"
    if [ -e "$f" ]; then
      printf '%-8s %s  stat=%s\n' "${s:-haupt}" "$(shasum -a 256 "$f" | cut -c1-64)" "$(stat -f '%i %z %m' "$f")"
    else
      printf '%-8s fehlt\n' "${s:-haupt}"
    fi
  done
  echo "Ordner: $(ls -1A "$(dirname "$2")" | tr '\n' ' ')"
}
vergleich() { # $1 $2 = zwei Zustandsdateien
  if diff <(tail -n +2 "$1") <(tail -n +2 "$2") >/dev/null; then echo "  => identisch"
  else echo "  => verschieden:"; diff <(tail -n +2 "$1") <(tail -n +2 "$2") | sed 's/^/     /'; fi
}
lesen() { # $1 = cli|lib, $2 = URI-Parameter, $3 = Pfad
  if [ "$1" = cli ]; then out=$($SQ "file:$3?$2" 'select count(*) from t;' 2>&1); rc=$?
  else out=$("$H/leser" "file:$3?$2" 2>&1); rc=$?; fi
  echo "Leser $1 ($2): Ausgabe=[$out] exit=$rc"
}

echo "=== (a) Bibliothek"
echo "\$ otool -L $SQ"; otool -L $SQ
echo "\$ $SQ --version"; $SQ --version
echo "\$ otool -L leser"; otool -L "$H/leser"
echo "\$ leser --version (sqlite3_libversion, sqlite3_sourceid aus /usr/lib/libsqlite3.dylib)"; "$H/leser" --version

for L in cli lib; do
echo
echo "################ Leser: $L"
echo
echo "=== (b/$L) WAL-Datenbank, Schreiber laeuft und haelt eine Zeile in -wal"
D=$W/b-$L; mkdir "$D"; DB=$D/x.db
$SQ "$DB" "pragma journal_mode=wal; create table t(x); insert into t values(1),(2),(3);" >/dev/null
FIFO=$H/fifo-$L; mkfifo "$FIFO"; LOG=$H/schreiber-$L.log
$SQ "$DB" < "$FIFO" > "$LOG" 2>&1 &
SPID=$!
exec 3> "$FIFO"
printf '%s\n' "pragma wal_autocheckpoint=0;" "insert into t values(4);" ".print BEREIT" >&3
for i in $(seq 1 50); do grep -q BEREIT "$LOG" && break; sleep 0.1; done
echo "Schreiber pid=$SPID, Protokoll: $(tr '\n' ' ' < "$LOG")"
sleep 1.1; zustand "b/$L: Kontrolle 1, ohne Leser" "$DB" > $H/b0; cat $H/b0
sleep 1.1; zustand "b/$L: Kontrolle 2, ohne Leser, 1 s spaeter" "$DB" > $H/b1; cat $H/b1
echo "Kontrolle 1 gegen Kontrolle 2 (der ruhende Schreiber allein):"; vergleich $H/b0 $H/b1
sleep 1.1; lesen $L "mode=ro" "$DB"
zustand "b/$L: nach dem Leser (Leser geschlossen, Schreiber offen)" "$DB" > $H/b2; cat $H/b2
echo "Kontrolle 2 gegen nach dem Leser:"; vergleich $H/b1 $H/b2
sleep 1.1; lesen $L "mode=ro" "$DB"; lesen $L "mode=ro" "$DB"
zustand "b/$L: nach zwei weiteren Lesern" "$DB" > $H/b3; cat $H/b3
echo "nach dem ersten Leser gegen nach zwei weiteren:"; vergleich $H/b2 $H/b3

echo
echo "=== (c/$L) dieselbe Datenbank nach dem Ende des Schreibers (kill -9, ohne Schliessen; -wal und -shm bleiben)"
kill -9 $SPID; wait $SPID 2>/dev/null; exec 3>&-
sleep 1.1; zustand "c/$L: Schreiber beendet, Kontrolle 1" "$DB" > $H/c0; cat $H/c0
sleep 1.1; zustand "c/$L: Kontrolle 2, 1 s spaeter" "$DB" > $H/c1; cat $H/c1
echo "Kontrolle 1 gegen Kontrolle 2:"; vergleich $H/c0 $H/c1
sleep 1.1; lesen $L "mode=ro" "$DB"
zustand "c/$L: nach dem Leser (er war die einzige und damit letzte Verbindung)" "$DB" > $H/c2; cat $H/c2
echo "Kontrolle 2 gegen nach dem Leser:"; vergleich $H/c1 $H/c2
sleep 1.1; lesen $L "mode=ro" "$DB"
zustand "c/$L: nach einem zweiten Leser" "$DB" > $H/c3; cat $H/c3
echo "nach dem ersten Leser gegen nach dem zweiten:"; vergleich $H/c2 $H/c3

echo
echo "=== (d/$L) WAL-Datenbank ohne -wal und -shm"
D=$W/d-$L; mkdir "$D"; DB=$D/x.db
$SQ "$DB" "pragma journal_mode=wal; create table t(x); insert into t values(1),(2),(3);" >/dev/null
echo "Nach dem sauberen Schliessen des Anlegers: $(ls -1A "$D" | tr '\n' ' ')(Groesse -wal: $(stat -f %z "$DB-wal" 2>/dev/null || echo fehlt))"
echo "-wal und -shm werden entfernt (die Datenbank ist zurueckgeschrieben, -wal ist leer)."
rm -f "$DB-wal" "$DB-shm"
echo "Kopfbytes 18/19 (02 02 = WAL): $(xxd -s 18 -l 2 -p "$DB")"
sleep 1.1; zustand "d/$L: vor den Lesern" "$DB" > $H/d1; cat $H/d1
lesen $L "mode=ro" "$DB"
zustand "d/$L: nach mode=ro" "$DB" > $H/d2; cat $H/d2
echo "vor gegen nach mode=ro:"; vergleich $H/d1 $H/d2
lesen $L "mode=ro&immutable=1" "$DB"
zustand "d/$L: nach immutable=1" "$DB" > $H/d3; cat $H/d3
echo "vor gegen nach immutable=1:"; vergleich $H/d1 $H/d3

echo
echo "=== (e/$L) Rollback-Datenbank mit mode=ro"
D=$W/e-$L; mkdir "$D"; DB=$D/x.db
$SQ "$DB" "pragma journal_mode=delete; create table t(x); insert into t values(1),(2),(3);" >/dev/null
echo "Kopfbytes 18/19 (01 01 = Rollback): $(xxd -s 18 -l 2 -p "$DB")"
sleep 1.1; zustand "e/$L: vor dem Leser" "$DB" > $H/e1; cat $H/e1
lesen $L "mode=ro" "$DB"
zustand "e/$L: nach dem Leser" "$DB" > $H/e2; cat $H/e2
echo "vor gegen nach:"; vergleich $H/e1 $H/e2
done

rm -rf "$W" "$H"

/* Wegwerf-Leser: oeffnet eine URI nur lesend ueber /usr/lib/libsqlite3.dylib,
   liest count(*) aus t und schliesst. Mit "--version" nur die Bibliotheksfassung. */
#include <stdio.h>
#include <string.h>
#include <sqlite3.h>
int main(int argc, char **argv) {
  if (argc == 2 && strcmp(argv[1], "--version") == 0) {
    printf("%s %s\n", sqlite3_libversion(), sqlite3_sourceid());
    return 0;
  }
  sqlite3 *db = NULL;
  int rc = sqlite3_open_v2(argv[1], &db, SQLITE_OPEN_READONLY | SQLITE_OPEN_URI, NULL);
  if (rc != SQLITE_OK) { printf("open: %d %s\n", sqlite3_extended_errcode(db), sqlite3_errmsg(db)); sqlite3_close_v2(db); return 1; }
  sqlite3_stmt *st = NULL;
  rc = sqlite3_prepare_v2(db, "select count(*) from t", -1, &st, NULL);
  if (rc != SQLITE_OK) { printf("prepare: %d %s\n", sqlite3_extended_errcode(db), sqlite3_errmsg(db)); sqlite3_close_v2(db); return 1; }
  rc = sqlite3_step(st);
  if (rc == SQLITE_ROW) printf("%lld\n", sqlite3_column_int64(st, 0));
  else printf("step: %d %s\n", sqlite3_extended_errcode(db), sqlite3_errmsg(db));
  sqlite3_finalize(st);
  int c = sqlite3_close_v2(db);
  if (c != SQLITE_OK) { printf("close: %d\n", c); return 1; }
  return rc == SQLITE_ROW ? 0 : 1;
}
